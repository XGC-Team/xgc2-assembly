cmake_minimum_required(VERSION 3.16)
project(xgc2_assembly VERSION 0.1.0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The initial loader is supported on Linux only")
endif()
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(XA_SANITIZE_ADDRESS "Enable AddressSanitizer and UBSan" OFF)
option(XA_SANITIZE_THREAD "Enable ThreadSanitizer" OFF)
if(XA_SANITIZE_ADDRESS AND XA_SANITIZE_THREAD)
  message(FATAL_ERROR "Select one sanitizer configuration")
endif()
if(XA_SANITIZE_ADDRESS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(XA_SANITIZE_THREAD)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(xgc2_assembly STATIC src/assembly.cpp src/domain_thread.cpp)
add_library(xgc2::assembly ALIAS xgc2_assembly)
set_target_properties(xgc2_assembly PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME assembly)
target_include_directories(xgc2_assembly PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(xgc2_assembly PUBLIC cxx_std_17)
target_link_libraries(xgc2_assembly PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(xgc2_assembly PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(xa_scalar MODULE examples/scalar.c)
target_include_directories(xa_scalar PRIVATE include)
set_target_properties(xa_scalar PROPERTIES C_VISIBILITY_PRESET hidden)
target_compile_options(xa_scalar PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(xa_demo examples/demo.cpp)
target_link_libraries(xa_demo PRIVATE xgc2::assembly)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS xgc2_assembly EXPORT xgc2_assemblyTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT xgc2_assemblyTargets NAMESPACE xgc2:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xgc2_assembly)
configure_package_config_file(cmake/xgc2_assemblyConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/xgc2_assemblyConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xgc2_assembly)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/xgc2_assemblyConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/xgc2_assemblyConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/xgc2_assemblyConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xgc2_assembly)
